#!/usr/bin/env bash
set -euo pipefail
left=$1
right=$2
rounds=$3
repeat=$4
out=$5
for ((round=1; round<=rounds; round++)); do
    if ((round % 2)); then order='A B'; else order='B A'; fi
    for side in $order; do
        if [[ $side == A ]]; then bin=$left; else bin=$right; fi
        printf 'ROUND %d %s\n' "$round" "$side" >> "$out"
        "$bin" --repeat "$repeat" >> "$out"
    done
done
